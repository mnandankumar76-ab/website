<script>

function showPage(pageId, clickedLink) {

    // Hide all pages
    document.querySelectorAll('.page').forEach(page => {
        page.classList.remove('active');
    });

    // Display selected page
    document.getElementById(pageId).classList.add('active');

    // Remove active navigation style
    document.querySelectorAll('nav ul a').forEach(link => {
        link.classList.remove('active');
    });

    // Highlight selected navigation link
    if (clickedLink) {
        clickedLink.classList.add('active');
    } else {
        document.querySelectorAll('nav ul a').forEach(link => {
            if (link.getAttribute('href') === '#' + pageId) {
                link.classList.add('active');
            }
        });
    }

    // Update browser URL
    history.replaceState(null, '', '#' + pageId);

    // Scroll to top
    window.scrollTo(0, 0);
}

// Open correct page when using browser back/forward
window.addEventListener('hashchange', function() {
    const pageId = location.hash.substring(1);
    if (document.getElementById(pageId)?.classList.contains('page')) {
        showPage(pageId);
    }
});

// Open page from URL hash
window.addEventListener('DOMContentLoaded', function() {
    const pageId = location.hash.substring(1);
    if (document.getElementById(pageId)?.classList.contains('page')) {
        showPage(pageId);
    }
});

</script>
